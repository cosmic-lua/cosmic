note: `require` is a special function, so a call of it is checked by its handler and 53b never sees its argument: `require(nil)` and `require(name)` with `name: string | nil` compiled and raised at run time. The handler refuses an argument that is nil or may be, as 53b refuses one for any parameter. Any other argument that is not a string literal is typed `any`
--- find
         if node.e2[1].kind ~= "string" then
            return a_tuple(node, { a_type(node, "any", {}) })
         end
--- replace
         local named = b.tuple[1]
         if named and (named.typename == "nil" or without_nil(self, node, named, false) ~= nil) then
            return self.errs:invalid_at(node.e2[1], "require: argument 1: got %s, expected string", named)
         end
         if node.e2[1].kind ~= "string" then
            return a_tuple(node, { a_type(node, "any", {}) })
         end
--- end
