function Errors:invalid_key_at(w: Where, fields: {string}, msg: string, ...:Type): InvalidType
   local before = #self.errors
   self:add(w, msg, ...)
   if fields and #self.errors > before then
      self.errors[#self.errors].fields = fields
   end
   return an_invalid(w)
end
