note: two function types are the same type only when both or neither never return: this is the comparison of a generic type argument, an invariant parameter or return, and the type a record function declares against its field, so `function Guard.die(): never` is refused when the field `die` is declared without `: never`, and the reverse (43d5)

            local argdelta = a.is_method and 1 or 0
--- replace
         ["function"] = function(self:TypeChecker, a: FunctionType, b: FunctionType): boolean, {Error}
            if (not not a.never) ~= (not not b.never) then
               return false, { Err("got %s, expected %s, which differ in whether the function never returns", a, b) }
            end
            local argdelta = a.is_method and 1 or 0
--- end
