note: the type a record or interface field gets when it is declared `name?: T`: T united with nil, written as `T | nil` is, so narrowing, `is` and every comparison treat it as any other nil-admitting field. A union T already is gets nil as one more member rather than a second union around it, and one that has nil is left alone. A generic function type is refused (its message says a plain field can be omitted anyway): a union of a generic is no type the checker reads, and the declaration has no other place to put nil

local function optional_field_type(ps: ParseState, i: integer, t: Type): Type
   if t is GenericType then
      fail(ps, i, "an optional field cannot have a generic type (a plain field `f: function<T>(T): T` can be left out of a literal anyway)")
      return t
   end
   local members: {Type} = {}
   if t is UnionType then
      for _, m in ipairs(t.types) do
         table.insert(members, m)
      end
   else
      members[1] = t
   end
   for _, m in ipairs(members) do
      if m is NilType then
         return t
      end
   end
   table.insert(members, new_type(ps, i, "nil") as NilType)
   local u = new_type(ps, i, "union") as UnionType
   u.types = members
   return u
end
