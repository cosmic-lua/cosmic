         local ra, rb = a, b
         if ra is TypeVarType then
            ra = self:find_var_type(ra.typevar) or ra
         end
         if rb is TypeVarType then
            rb = self:find_var_type(rb.typevar) or rb
         end
         if ra is NominalType then
            ra = self:resolve_nominal(ra)
         end
         if rb is NominalType then
            rb = self:resolve_nominal(rb)
         end
         if ra is FunctionType and rb is FunctionType and rb.never and not ra.never then
            errs = { Err("got %s, expected %s: a function that may return does not stand for one that never returns", ra, rb) }

            if ok then
               return true
