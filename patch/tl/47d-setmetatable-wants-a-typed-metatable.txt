note: `setmetatable(t, mt)` is `function<T>(T, metatable<T>): T`, and while T is still a variable the parameter `metatable<T>` has no record to hold a metatable to (47a and the metamethod check of `check_metatable_contract` apply to `metatable<R>`), so a variable whose type is a record that merely looks like a metatable (`local mt = { __index = { name = 1 } }`) passed. A metatable that is not a table literal must now be typed `metatable<T>`, which `local mt: metatable<R> = {...}` holds to R's fields and metamethods where it is declared; a table literal in the call is held to them there as before. A variable of a record type is refused with that annotation as the fix
--- find
      ["pcall"] = special_pcall_xpcall,
      ["xpcall"] = special_pcall_xpcall,
--- replace
      ["setmetatable"] = function(self: TypeChecker, node: Node, a: Type, b: TupleType, argdelta?: integer): InvalidOrTupleType
         local r = self:type_check_function_call(node, a, b, argdelta)
         local arg = node.e2 and node.e2[2]
         local given = b.tuple[2]
         local structural = given and self:to_structural(given)
         if arg and arg.kind ~= "literal_table" and given and not (given is NominalType)
            and structural is RecordLikeType then
            self.errs:add(arg, "a metatable that is not written in the call must be typed metatable<T>, to be checked against T: declare it `local mt: metatable<T> = { ... }`")
         end
         return r
      end,
      ["pcall"] = special_pcall_xpcall,
      ["xpcall"] = special_pcall_xpcall,
--- end
