note: a field fact (`m.f ~= nil`) is applied against the type of its base `m`; the base is read without its nil member, since reading `m.f` proved `m` is not nil, so `if m ~= nil and m.f ~= nil` narrows `m.f` for a base that is itself a nil union
--- find
      local base = self:find_var_type(string.sub(f.var, 1, dot - 1), "check_only")
      if not base then
         return nil
      end
--- replace
      local base = self:find_var_type(string.sub(f.var, 1, dot - 1), "check_only")
      if not base then
         return nil
      end
      base = without_nil(self, f.w, base, false) or base
--- end
