note: this header is the first one nearly every file of the crypto
  library and the TLS layer includes, and on an Armv8 target it pulls
  in <arm_neon.h>, clang's header of NEON intrinsics, which is slow to
  parse. Only the files that use NEON intrinsics, or call
  mbedtls_xor() (whose fast path uses them), need it. A file compiled
  with -DCOSMIC_NO_NEON_HEADER leaves it out, and with it
  MBEDTLS_HAVE_NEON_INTRINSICS; build.zig passes the macro only to
  files whose object is the same either way. Without the macro this is
  upstream's header.

#include <arm_neon.h>
#define MBEDTLS_HAVE_NEON_INTRINSICS
#elif defined(MBEDTLS_PLATFORM_IS_WINDOWS_ON_ARM64)
