/*
** Whether node iNode is in pCheck's set of nodes the walk has read, and,
** if bAdd, put it there. A node number below 1 names no node and is
** never in the set.
*/
static int rtreeCheckSeen(RtreeCheck *pCheck, i64 iNode, int bAdd){
  sqlite3_uint64 mask, h;
  if( iNode<1 || pCheck->aSeen==0 ) return 0;
  mask = (sqlite3_uint64)pCheck->nSeenSlot - 1;
  h = ((sqlite3_uint64)iNode * 0x9E3779B97F4A7C15ULL) & mask;
  while( pCheck->aSeen[h] ){
    if( pCheck->aSeen[h]==iNode ) return 1;
    h = (h+1) & mask;
  }
  if( bAdd ) pCheck->aSeen[h] = iNode;
  return 0;
}
