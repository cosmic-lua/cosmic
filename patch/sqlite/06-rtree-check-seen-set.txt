/*
** The slot of aSeen, which has 1<<nBit slots, that holds iNode, or the
** empty slot where iNode would go.
*/
static i64 *rtreeCheckSlot(i64 *aSeen, int nBit, i64 iNode){
  sqlite3_uint64 mask = ((sqlite3_uint64)1<<nBit) - 1;
  sqlite3_uint64 h = ((sqlite3_uint64)iNode * 0x9E3779B97F4A7C15ULL) >> (64-nBit);
  while( aSeen[h] && aSeen[h]!=iNode ){
    h = (h+1) & mask;
  }
  return &aSeen[h];
}

/*
** Whether node iNode, 1 or above, is in pCheck's set of nodes the walk
** has read.
*/
static int rtreeCheckSeen(RtreeCheck *pCheck, i64 iNode){
  assert( iNode>=1 );
  if( pCheck->aSeen==0 ) return 0;
  return *rtreeCheckSlot(pCheck->aSeen, pCheck->nSeenBit, iNode)!=0;
}

/*
** Put node iNode, 1 or above and not yet in the set, in pCheck's set of
** nodes the walk has read, doubling the set first if the add would fill
** more than half its slots.
*/
static void rtreeCheckAddSeen(RtreeCheck *pCheck, i64 iNode){
  assert( iNode>=1 );
  if( pCheck->aSeen==0 || 2*(pCheck->nSeen+1)>((i64)1<<pCheck->nSeenBit) ){
    int nBit = pCheck->aSeen ? pCheck->nSeenBit+1 : 4;
    i64 nOld = pCheck->aSeen ? ((i64)1<<pCheck->nSeenBit) : 0;
    i64 *aNew = sqlite3_malloc64(sizeof(i64)*((i64)1<<nBit));
    i64 i;
    if( aNew==0 ){
      pCheck->rc = SQLITE_NOMEM;
      return;
    }
    memset(aNew, 0, sizeof(i64)*((i64)1<<nBit));
    for(i=0; i<nOld; i++){
      if( pCheck->aSeen[i] ){
        *rtreeCheckSlot(aNew, nBit, pCheck->aSeen[i]) = pCheck->aSeen[i];
      }
    }
    sqlite3_free(pCheck->aSeen);
    pCheck->aSeen = aNew;
    pCheck->nSeenBit = nBit;
  }
  *rtreeCheckSlot(pCheck->aSeen, pCheck->nSeenBit, iNode) = iNode;
  pCheck->nSeen++;
}
