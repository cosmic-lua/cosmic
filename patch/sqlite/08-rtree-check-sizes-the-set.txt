note: Before the walk, the check counts the rows of %_node and makes the set (05-rtree-check-reads-each-node-once.txt) a power of two at least twice that count. A count that fails leaves its error in the check's return code; a count past 2^39 rows, or a set that cannot be had, fails the check as out of memory; in either case the walk does not run.

    if( check.rc==SQLITE_OK ){
      sqlite3_stmt *pCount = rtreeCheckPrepare(&check,
          "SELECT count(*) FROM %Q.'%q_node'", zDb, zTab
      );
      if( pCount ){
        if( sqlite3_step(pCount)==SQLITE_ROW ){
          check.nNode = sqlite3_column_int64(pCount, 0);
        }
        check.rc = sqlite3_finalize(pCount);
      }
    }
    if( check.rc==SQLITE_OK ){
      if( check.nNode>((i64)1<<39) ){
        check.rc = SQLITE_NOMEM;
      }else{
        check.nSeenSlot = 16;
        while( check.nSeenSlot<2*check.nNode ) check.nSeenSlot *= 2;
        check.aSeen = sqlite3_malloc64(sizeof(i64)*check.nSeenSlot);
        if( check.aSeen==0 ){
          check.rc = SQLITE_NOMEM;
        }else{
          memset(check.aSeen, 0, sizeof(i64)*check.nSeenSlot);
        }
      }
    }
    if( check.rc==SQLITE_OK ){
      rtreeCheckNode(&check, 0, 0, 1);
    }
