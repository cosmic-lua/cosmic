note: tointeger() and toreal() parse text with totype.c's own copies of SQLite's parsers, which drift from SQLite's: toreal('1.7976931348623157e308') is Inf, reals across the range come out a unit in the last place off ('-1.0e-290'), and tointeger reads '12\0x' as 12, stopping at the NUL. This helper reads a text the way CAST and column affinity do, and only when the whole text is a number; the next two records use it in place of the copies.

/*
** The number the whole of a text spells, read as SQLite reads it (as CAST
** and column affinity do), as a new value holding an integer or a real.
** NULL, with no error, when the text has a NUL byte, leading or trailing
** space, or anything else that is not part of one number; NULL with the
** context's out-of-memory error when the copy cannot be made. The caller
** frees what it is given.
*/
static sqlite3_value *totypeNumber(sqlite3_context *context, sqlite3_value *pText){
  const unsigned char *z = sqlite3_value_text(pText);
  int n = sqlite3_value_bytes(pText);
  sqlite3_value *pNum;
  if( z==0 || n==0 || totypeIsspace(z[0]) || totypeIsspace(z[n-1]) ) return 0;
  if( memchr(z, 0, n) ) return 0;
  pNum = sqlite3_value_dup(pText);
  if( pNum==0 ){
    sqlite3_result_error_nomem(context);
    return 0;
  }
  switch( sqlite3_value_numeric_type(pNum) ){
    case SQLITE_INTEGER:
    case SQLITE_FLOAT:
      return pNum;
  }
  sqlite3_value_free(pNum);
  return 0;
}
