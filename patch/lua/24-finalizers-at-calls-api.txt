--- find
    case LUA_GCCOLLECT: {
      luaC_fullgc(L, 0);
      break;
    }
--- replace
    case LUA_GCCOLLECT: {
      luaC_fullgc(L, 0);
      luaC_runfinalizers(L);
      break;
    }
--- end
