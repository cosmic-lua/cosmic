/*
** The class literal: 'constructor' has made the table in the register
** 'v' stands for; attach the class table the path after '@' names.
*/
static void classliteral (LexState *ls, expdesc *v) {
  /* classliteral -> constructor '@' NAME { '.' NAME } */
  FuncState *fs = ls->fs;
  expdesc mt;
  int table = v->u.info;
  luaX_next(ls);  /* skip '@' */
  singlevar(ls, &mt);
  while (ls->t.token == '.')
    fieldsel(ls, &mt);
  luaK_exp2anyreg(fs, &mt);
  luaK_codeABC(fs, OP_SETCLASS, table, mt.u.info, 0);
  fs->freereg = cast_byte(table + 1);  /* free the register of the path */
}
