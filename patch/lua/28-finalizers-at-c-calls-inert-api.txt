LUA_API int lua_setinert (lua_State *L, int idx) {
  const TValue *o;
  int done;
  lua_lock(L);
  o = index2value(L, idx);
  if (ttislcf(o))
    done = luaC_setinert(fvalue(o));
  else if (ttisCclosure(o))
    done = luaC_setinert(clCvalue(o)->f);
  else
    done = 1;
  lua_unlock(L);
  return done;
}
