LUA_API int lua_setinert (lua_State *L, int idx) {
  const TValue *o;
  int done;
  lua_lock(L);
  o = index2value(L, idx);
  done = !ttislcf(o) || luaC_setinert(fvalue(o));
  lua_unlock(L);
  return done;
}
