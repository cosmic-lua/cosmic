/*
** Call the pending finalizers, where the program calls a Lua function
** or collectgarbage: never inside a collection, an emergency one, or
** a finalizer, and not while the state closes (it calls them itself).
*/
void luaC_runfinalizers (lua_State *L) {
  global_State *g = G(L);
  while (g->tobefnz != NULL && !g->gcemergency &&
         !(g->gcstp & (GCSTPGC | GCSTPCLS)) && luaD_checkminstack(L))
    GCTM(L);
}
