#define INERT_MOST 128

static lua_CFunction inert[INERT_MOST];
static int inert_count = 0;


/* Whether 'f' is a C function lua_setinert named. */
int luaC_inert (lua_CFunction f) {
  int i;
  for (i = 0; i < inert_count; i++)
    if (inert[i] == f)
      return 1;
  return 0;
}


/* Names 'f' a C function whose calls run no pending finalizer; 0 when
** the list is full. */
int luaC_setinert (lua_CFunction f) {
  if (luaC_inert(f))
    return 1;
  if (inert_count == INERT_MOST)
    return 0;
  inert[inert_count++] = f;
  return 1;
}
